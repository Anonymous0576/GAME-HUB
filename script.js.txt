const modal = document.getElementById("gameModal");
const container = document.getElementById("gameContainer");

let selectedGame = null;
let gameInterval = null;


/* =========================================
   OPEN GAME
========================================= */

function openGame(game) {

    selectedGame = game;

    const titles = {

        snake: "Snake Classic",

        space: "Space Blaster",

        racing: "Neon Racer",

        memory: "Memory Master",

        chess: "Mini Chess",

        bird: "Flappy Bird"

    };

    document.getElementById("gameTitle").textContent =
        titles[game] || "Game";

    document.getElementById("gameDescription").textContent =
        "Browser game";

    modal.classList.add("show");

    container.innerHTML = `

        <div class="game-start-screen">

            <div class="big-game-icon">
                🎮
            </div>

            <h2>
                ${titles[game] || "Game"}
            </h2>

            <p>
                Ready to play?
            </p>

            <button onclick="launchSelectedGame()">
                START GAME
            </button>

        </div>

    `;

}


/* =========================================
   CLOSE GAME
========================================= */

function closeGame() {

    modal.classList.remove("show");

    stopGame();

}


/* =========================================
   START SELECTED GAME
========================================= */

function launchSelectedGame() {

    stopGame();

    if (selectedGame === "snake") {
        startSnake();
    }

    else if (selectedGame === "space") {
        startSpace();
    }

    else if (selectedGame === "racing") {
        startRacing();
    }

    else if (selectedGame === "memory") {
        startMemory();
    }

    else {
        startComingSoon();
    }

}


/* =========================================
   STOP GAME
========================================= */

function stopGame() {

    if (gameInterval) {

        clearInterval(gameInterval);

        gameInterval = null;

    }

}


/* =========================================
   RESTART
========================================= */

function restartGame() {

    if (!selectedGame)
        return;

    launchSelectedGame();

}


/* =========================================
   FULLSCREEN
========================================= */

function fullscreenGame() {

    const windowElement =
        document.querySelector(".game-window");

    if (!document.fullscreenElement) {

        windowElement.requestFullscreen();

    } else {

        document.exitFullscreen();

    }

}

const searchInput =
    document.getElementById("searchInput");

searchInput.addEventListener("input", function () {

    const query =
        this.value.toLowerCase().trim();

    document
        .querySelectorAll(".game-card")
        .forEach(card => {

            const name =
                card.dataset.name.toLowerCase();

            if (name.includes(query)) {

                card.style.display = "";

            } else {

                card.style.display = "none";

            }

        });

});


/* CTRL + K SEARCH */

document.addEventListener("keydown", function(event) {

    if (
        event.ctrlKey &&
        event.key.toLowerCase() === "k"
    ) {

        event.preventDefault();

        searchInput.focus();

    }

});

function startSnake() {

    container.innerHTML = `
        <canvas
            id="gameCanvas"
            width="600"
            height="500">
        </canvas>
    `;

    const canvas =
        document.getElementById("gameCanvas");

    const ctx =
        canvas.getContext("2d");

    const size = 25;

    let snake = [
        {x: 300, y: 250},
        {x: 275, y: 250},
        {x: 250, y: 250}
    ];

    let dx = size;
    let dy = 0;

    let food = randomFood();

    let score = 0;


    function randomFood() {

        return {

            x:
                Math.floor(
                    Math.random() *
                    (canvas.width / size)
                ) * size,

            y:
                Math.floor(
                    Math.random() *
                    (canvas.height / size)
                ) * size

        };

    }


    function draw() {

        ctx.fillStyle = "#050912";

        ctx.fillRect(
            0,
            0,
            canvas.width,
            canvas.height
        );


        const head = {

            x: snake[0].x + dx,

            y: snake[0].y + dy

        };


        if (

            head.x < 0 ||
            head.y < 0 ||
            head.x >= canvas.width ||
            head.y >= canvas.height ||

            snake.some(
                part =>
                    part.x === head.x &&
                    part.y === head.y
            )

        ) {

            alert("Game Over! Score: " + score);

            startSnake();

            return;

        }


        snake.unshift(head);


        if (
            head.x === food.x &&
            head.y === food.y
        ) {

            score++;

            food = randomFood();

        }

        else {

            snake.pop();

        }


        /* FOOD */

        ctx.fillStyle = "#ff4d6d";

        ctx.fillRect(
            food.x,
            food.y,
            size - 2,
            size - 2
        );


        /* SNAKE */

        ctx.fillStyle = "#7c5cff";

        snake.forEach(part => {

            ctx.fillRect(
                part.x,
                part.y,
                size - 2,
                size - 2
            );

        });


        ctx.fillStyle = "white";

        ctx.font = "18px Arial";

        ctx.fillText(
            "Score: " + score,
            15,
            25
        );

    }


    function key(event) {

        if (
            event.key === "ArrowUp" &&
            dy === 0
        ) {

            dx = 0;
            dy = -size;

        }

        if (
            event.key === "ArrowDown" &&
            dy === 0
        ) {

            dx = 0;
            dy = size;

        }

        if (
            event.key === "ArrowLeft" &&
            dx === 0
        ) {

            dx = -size;
            dy = 0;

        }

        if (
            event.key === "ArrowRight" &&
            dx === 0
        ) {

            dx = size;
            dy = 0;

        }

    }


    document.onkeydown = key;

    gameInterval =
        setInterval(draw, 100);

}


/* =========================================
   SPACE SHOOTER
========================================= */

function startSpace() {

    container.innerHTML = `

        <canvas
            id="gameCanvas"
            width="800"
            height="500">
        </canvas>

    `;

    const canvas =
        document.getElementById("gameCanvas");

    const ctx =
        canvas.getContext("2d");

    let player = {

        x: 380,

        y: 430,

        width: 40,

        height: 40

    };

    let bullets = [];

    let enemies = [];

    let score = 0;

    let keys = {};


    document.onkeydown = event => {

        keys[event.key] = true;

        if (event.key === " ") {

            bullets.push({

                x: player.x + 18,

                y: player.y

            });

        }

    };


    document.onkeyup = event => {

        keys[event.key] = false;

    };


    for (let i = 0; i < 8; i++) {

        enemies.push({

            x: 50 + i * 95,

            y: 60,

            size: 35,

            speed: 1 + Math.random()

        });

    }


    function draw() {

        ctx.fillStyle = "#030711";

        ctx.fillRect(
            0,
            0,
            canvas.width,
            canvas.height
        );


        /* PLAYER */

        if (keys["ArrowLeft"])
            player.x -= 5;

        if (keys["ArrowRight"])
            player.x += 5;


        player.x =
            Math.max(
                0,
                Math.min(
                    canvas.width - player.width,
                    player.x
                )
            );


        ctx.fillStyle = "#7c5cff";

        ctx.fillRect(
            player.x,
            player.y,
            player.width,
            player.height
        );


        /* BULLETS */

        ctx.fillStyle = "#38bdf8";

        bullets.forEach(bullet => {

            bullet.y -= 8;

            ctx.fillRect(
                bullet.x,
                bullet.y,
                5,
                15
            );

        });


        /* ENEMIES */

        ctx.fillStyle = "#ff4d6d";

        enemies.forEach(enemy => {

            enemy.y += enemy.speed;

            ctx.fillRect(
                enemy.x,
                enemy.y,
                enemy.size,
                enemy.size
            );

        });


        /* COLLISION */

        bullets.forEach((bullet, bi) => {

            enemies.forEach((enemy, ei) => {

                if (

                    bullet.x < enemy.x + enemy.size &&
                    bullet.x + 5 > enemy.x &&
                    bullet.y < enemy.y + enemy.size &&
                    bullet.y + 15 > enemy.y

                ) {

                    bullets.splice(bi, 1);

                    enemies.splice(ei, 1);

                    score++;

                    enemies.push({

                        x: Math.random() *
                            (canvas.width - 40),

                        y: -40,

                        size: 35,

                        speed: 1 + Math.random() * 2

                    });

                }

            });

        });


        ctx.fillStyle = "white";

        ctx.font = "18px Arial";

        ctx.fillText(
            "Score: " + score,
            15,
            25
        );

    }


    gameInterval =
        setInterval(draw, 30);

}

            ${symbols.map((symbol,index) => `

                    onclick="flipCard(${index}, '${symbol}')"
                    style="
                        width:90px;
                        height:90px;
                        font-size:35px;
                        background:#151e2f;
                        border:1px solid #263248;
                        border-radius:10px;
                        color:white;
                    ">
                    ?
                </button>

            `).join("")}

        </div>

    `;

}


let firstCard = null;
let lockCards = false;


function flipCard(index, symbol) {

    if (lockCards)
        return;

    const card =
        document.getElementById(
            "card-" + index
        );

    card.textContent = symbol;


    if (firstCard === null) {

        firstCard = {
            index,
            symbol
        };

        return;

    }


    if (firstCard.symbol === symbol) {

        firstCard = null;

    } else {

        lockCards = true;

        setTimeout(() => {

            document.getElementById(
                "card-" + firstCard.index
            ).textContent = "?";

            card.textContent = "?";

            firstCard = null;

            lockCards = false;

        }, 700);

    }

}
